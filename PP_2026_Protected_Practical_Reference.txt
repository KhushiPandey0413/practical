PRACTICAL 1 — BIAS DETECTION
import pandas as pd
from sklearn.linear_model import LogisticRegression
from sklearn.model_selection import train_test_split
df=pd.DataFrame({"gender":["Male","Female"]*20,"score":[60,80,55,75,90,65,70,85]*5})
df["hired"]=((df.score+(df.gender=="Male")*8)>=70).astype(int)
df["gender_code"]=(df.gender=="Male").astype(int)
X,y=df[["gender_code","score"]],df.hired
Xtr,Xte,ytr,yte=train_test_split(X,y,random_state=42)
m=LogisticRegression().fit(Xtr,ytr); t=Xte.copy(); t["pred"]=m.predict(Xte)
male=t.loc[t.gender_code==1,"pred"].mean(); female=t.loc[t.gender_code==0,"pred"].mean()
di=female/male if male else 0
print("Disparate Impact:",di,"\nStatistical Parity Difference:",female-male)
print("Potentially biased" if di<0.8 else "DI rule does not flag bias")

PRACTICAL 2 — PROPHET VS ARIMA
import numpy as np,pandas as pd,matplotlib.pyplot as plt
from prophet import Prophet
from statsmodels.tsa.arima.model import ARIMA
from sklearn.metrics import mean_absolute_percentage_error,mean_squared_error
np.random.seed(1)
df=pd.DataFrame({"Date":pd.date_range("2025-01-01",periods=120),"Sales":100+np.arange(120)*.5+np.random.randn(120)*5})
tr,te=df[:-30],df[-30:]
p=Prophet(weekly_seasonality=True,daily_seasonality=True).fit(tr.rename(columns={"Date":"ds","Sales":"y"}))
pf=p.predict(p.make_future_dataframe(periods=30)).tail(30)
a=ARIMA(tr.Sales,order=(2,1,1)).fit().forecast(30)
for n,v in [("Prophet",pf.yhat.values),("ARIMA",a.values)]:
 print(n,"MAPE",mean_absolute_percentage_error(te.Sales,v)*100,"RMSE",mean_squared_error(te.Sales,v)**.5)
plt.plot(te.Date,te.Sales,label="Actual"); plt.plot(te.Date,pf.yhat,label="Prophet"); plt.plot(te.Date,a,label="ARIMA")
plt.fill_between(te.Date,pf.yhat_lower,pf.yhat_upper,alpha=.2); plt.legend(); plt.show()

PRACTICAL 3 — CHURN, FN COST Rs. 10,000
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.compose import ColumnTransformer
from sklearn.preprocessing import OneHotEncoder,StandardScaler
from sklearn.pipeline import make_pipeline
from sklearn.impute import SimpleImputer
from sklearn.linear_model import LogisticRegression
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import confusion_matrix,accuracy_score,precision_score,recall_score
df=pd.DataFrame({"tenure":[2,5,10,20,40,60,3,8,15,50,4,30]*5,"charges":[90,80,70,60,50,40,100,95,75,45,110,55]*5,"contract":["Month","Month","Month","Year","Year","Two","Month","Month","Year","Two","Month","Year"]*5,"Churn":[1,1,1,0,0,0,1,1,0,0,1,0]*5})
X,y=df.drop(columns="Churn"),df.Churn
prep=ColumnTransformer([("num",make_pipeline(SimpleImputer(),StandardScaler()),["tenure","charges"]),("cat",OneHotEncoder(handle_unknown="ignore"),["contract"])])
Xtr,Xte,ytr,yte=train_test_split(X,y,test_size=.25,random_state=1,stratify=y)
for n,model in [("Logistic",LogisticRegression()),("Random Forest",RandomForestClassifier(random_state=1))]:
 pipe=make_pipeline(prep,model).fit(Xtr,ytr); pred=pipe.predict(Xte); cm=confusion_matrix(yte,pred,labels=[0,1]); fn=cm[1,0]
 print(n,"Accuracy",accuracy_score(yte,pred),"Precision",precision_score(yte,pred,zero_division=0),"Recall",recall_score(yte,pred,zero_division=0),"CM",cm,"FN Cost Rs.",fn*10000)

import pandas as pd
from surprise import Dataset, Reader, SVD, KNNBasic, accuracy
from surprise.model_selection import train_test_split

# Load dataset given by mam
df = pd.read_csv("ratings.csv")

# Prepare dataset
reader = Reader(rating_scale=(1, 5))
data = Dataset.load_from_df(
    df[["userId", "movieId", "rating"]], reader
)

# Split dataset
train, test = train_test_split(
    data, test_size=0.25, random_state=42
)

# Compare SVD and KNN
for name, model in [
    ("SVD", SVD(random_state=42)),
    ("KNN", KNNBasic(verbose=False))
]:
    model.fit(train)
    predictions = model.test(test)

    print(name)
    accuracy.rmse(predictions)
    accuracy.mae(predictions)

PRACTICAL 7 — RFM + K-MEANS
import pandas as pd,numpy as np
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans
np.random.seed(1)
df=pd.DataFrame({"CustomerID":np.random.randint(1,21,100),"Date":pd.to_datetime("2026-01-01")+pd.to_timedelta(np.random.randint(0,90,100),unit="D"),"Amount":np.random.randint(100,2000,100)})
today=df.Date.max()+pd.Timedelta(days=1)
rfm=df.groupby("CustomerID").agg(Recency=("Date",lambda x:(today-x.max()).days),Frequency=("Date","count"),Monetary=("Amount","sum")).reset_index()
rfm["Cluster"]=KMeans(n_clusters=3,random_state=1,n_init=10).fit_predict(StandardScaler().fit_transform(rfm[["Recency","Frequency","Monetary"]]))
print(rfm)

PRACTICAL 10 — APRIORI
import pandas as pd
from mlxtend.preprocessing import TransactionEncoder
from mlxtend.frequent_patterns import apriori,association_rules
transactions=[["Bread","Milk"],["Bread","Butter"],["Bread","Milk","Butter"],["Milk","Butter"],["Bread","Milk"],["Bread","Butter"]]
enc=TransactionEncoder(); basket=pd.DataFrame(enc.fit(transactions).transform(transactions),columns=enc.columns_)
items=apriori(basket,min_support=.3,use_colnames=True); print(items)
print(association_rules(items,metric="confidence",min_threshold=.4)[["antecedents","consequents","support","confidence","lift"]])
